"use client";
import { useEffect } from "react";

const isChunkError = (e: Error) =>
  /ChunkLoadError|Loading chunk|dynamically imported module|Importing a module script failed/i.test(`${e.name} ${e.message}`);

// Trang gặp lỗi: cho người dùng bấm thử lại / tải lại, không bị kẹt ở màn hình lỗi.
export default function ErrorPage({ error, reset }: { error: Error & { digest?: string }; reset: () => void }) {
  useEffect(() => {
    // Lỗi thiếu file JS thường do web vừa cập nhật mà trình duyệt còn giữ bản cũ: tự tải lại đúng 1 lần.
    if (!isChunkError(error)) return;
    try {
      if (sessionStorage.getItem("chunk-reload")) return;
      sessionStorage.setItem("chunk-reload", "1");
    } catch { return; }
    window.location.reload();
  }, [error]);

  return (
    <main className="mx-auto grid max-w-xl place-items-center px-4 py-32 text-center">
      <h1 className="font-display text-4xl font-extrabold tracking-tight sm:text-5xl">Trang gặp lỗi</h1>
      <p className="mt-4 text-haze">Có thể web vừa được cập nhật hoặc mạng bị gián đoạn. Thử lại, nếu vẫn lỗi thì tải lại cả trang.</p>
      <div className="mt-8 flex flex-wrap justify-center gap-3">
        <button onClick={reset}
                className="rounded-full bg-berry px-7 py-3.5 font-semibold text-ink transition hover:-translate-y-0.5 active:scale-95">
          Thử lại
        </button>
        <button onClick={() => window.location.reload()}
                className="glass rounded-full px-7 py-3.5 font-semibold transition hover:bg-white/15 active:scale-95">
          Tải lại trang
        </button>
      </div>
    </main>
  );
}

export async function generateMetadata({ params }: { params: Promise<{ slug: string }> }): Promise<Metadata> {
  const { slug } = await params;
  const game = games.find((g) => g.slug === slug);
  if (!game) return {};
  return {
    title: game.title,
    description: game.description,
    alternates: { canonical: `/game/${game.slug}` },
    openGraph: { title: game.title, description: game.description, type: "website", images: game.cover ? [game.cover] : undefined },
  };
}

import type { MetadataRoute } from "next";
import { games } from "@/data/games";
import { siteUrl } from "@/data/site";

// /sitemap.xml: trang chủ + mỗi game một dòng. lastModified lấy từ updatedAt của game nên Google biết trang nào mới đổi.
export default function sitemap(): MetadataRoute.Sitemap {
  if (!siteUrl) return []; // site.json -> url sai: bỏ qua thay vì làm build chết
  const latest = games.reduce((m, g) => (g.updatedAt > m ? g.updatedAt : m), "");
  return [
    { url: new URL("/", siteUrl).href, lastModified: latest || undefined, changeFrequency: "daily", priority: 1 },
    ...games.map((g) => ({
      url: new URL(`/game/${g.slug}`, siteUrl).href,
      lastModified: g.updatedAt,
      changeFrequency: "weekly" as const,
      priority: 0.7,
    })),
  ];
}

"use client";
import { useEffect, useRef, useState } from "react";
import { TURNSTILE_SITE_KEY } from "@/lib/supabase";

type TurnstileApi = {
  render: (el: HTMLElement, opts: Record<string, unknown>) => string;
  reset: (id?: string) => void;
  remove: (id?: string) => void;
};
declare global { interface Window { turnstile?: TurnstileApi } }

const SRC = "https://challenges.cloudflare.com/turnstile/v0/api.js?render=explicit";
let loader: Promise<void> | null = null;
const loadScript = () =>
  (loader ??= new Promise<void>((resolve, reject) => {
    if (window.turnstile) return resolve();
    const s = document.createElement("script");
    s.src = SRC; s.async = true; s.defer = true;
    s.onload = () => resolve();
    s.onerror = () => { loader = null; reject(new Error("turnstile")); };
    document.head.appendChild(s);
  }));

// Cloudflare Turnstile. onToken(token) khi qua xác minh, onToken(null) khi hết hạn/lỗi.
// Token chỉ dùng được MỘT lần: sau mỗi lần gửi form, cha tăng resetKey để lấy token mới.
export default function Turnstile({ onToken, resetKey = 0 }: { onToken: (t: string | null) => void; resetKey?: number }) {
  const box = useRef<HTMLDivElement>(null);
  const widget = useRef<string | null>(null);
  const cb = useRef(onToken);
  const [failed, setFailed] = useState(false);
  useEffect(() => { cb.current = onToken; });

  useEffect(() => {
    let dead = false;
    loadScript().then(
      () => {
        const el = box.current, api = window.turnstile;
        if (dead || !el || !api) return;
        widget.current = api.render(el, {
          sitekey: TURNSTILE_SITE_KEY,
          theme: "dark",
          size: "flexible",
          callback: (t: string) => cb.current(t),
          "expired-callback": () => cb.current(null),
          "error-callback": () => cb.current(null),
        });
      },
      () => { if (!dead) setFailed(true); },
    );
    return () => {
      dead = true;
      if (widget.current) { window.turnstile?.remove(widget.current); widget.current = null; }
    };
  }, []);

  useEffect(() => {
    if (widget.current) window.turnstile?.reset(widget.current);
  }, [resetKey]);

  if (failed) {
    return <p role="alert" className="rounded-2xl bg-red-500/10 px-4 py-3 text-sm text-red-300">
      Không tải được bước xác minh của Cloudflare. Tắt trình chặn quảng cáo cho trang này rồi tải lại.
    </p>;
  }
  return <div ref={box} className="min-h-[65px]" />;
}

  const safe = links.filter((l) => /^https?:\/\//i.test(l.url)); // chỉ nhận http/https

    if (!open) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape") { setZoom(false); setCur(null); }

"use client";
import { useMemo, useState, useSyncExternalStore } from "react";
import type { Game } from "@/data/games";
import { site } from "@/data/site";
import { applyFilters, buildTagIndex, parseFilters, serializeFilters, type Filters } from "@/lib/library-filter";
import FilterPanel from "./FilterPanel";
import GameCard from "./GameCard";
import { EmptyArt, GhostCard } from "./LibraryArt";

// Bộ lọc đồng bộ với URL (?engine=Unity&tag=Romance&mode=and&q=...&sort=name): chia sẻ được link, nút Back hoạt động.
// Đọc URL bằng useSyncExternalStore: HTML tĩnh vẫn chứa đủ game, client lọc sau khi hydrate.
const EVENT = "filter-change";
const subscribe = (cb: () => void) => {
  window.addEventListener("popstate", cb);
  window.addEventListener(EVENT, cb);
  return () => { window.removeEventListener("popstate", cb); window.removeEventListener(EVENT, cb); };
};
const getSearch = () => window.location.search;

// Tag gợi ý sửa trong src/data/site.json -> tagPresets (thiếu mục này thì chỉ hiện tag có trong games.json)
const PAGE = 30; // số game hiện mỗi lượt; bấm "Xem thêm" để hiện tiếp
const PRESETS = (site as { tagPresets?: string[] }).tagPresets ?? [];

export default function Library({ games }: { games: Game[] }) {
  const engines = useMemo(() => Array.from(new Set(games.map((g) => g.engine))), [games]);
  const tagIndex = useMemo(() => buildTagIndex(games, PRESETS), [games]);
  const search = useSyncExternalStore(subscribe, getSearch, () => "");
  const filters = useMemo(() => parseFilters(search, engines, tagIndex), [search, engines, tagIndex]);
  const list = useMemo(() => applyFilters(games, filters), [games, filters]);
  const [panelOpen, setPanelOpen] = useState(false); // chỉ dùng trên điện thoại/tablet
  // Số game đang hiện gắn với query hiện tại: đổi bộ lọc/sắp xếp thì tự về PAGE mà không cần effect.
  const [more, setMore] = useState({ key: "", n: PAGE });
  const limit = more.key === search ? more.n : PAGE;
  const shown = list.slice(0, limit);

  const dirty = filters.engine !== null || filters.q.trim() !== "" || filters.tags.length > 0;
  const badge = filters.tags.length + (filters.q.trim() ? 1 : 0);

  // replace=true cho ô gõ chữ: không đẩy mỗi phím vào lịch sử Back
  const update = (patch: Partial<Filters>, replace = false) => {
    const { pathname, hash } = window.location;
    const url = pathname + serializeFilters({ ...filters, ...patch }) + hash;
    window.history[replace ? "replaceState" : "pushState"](null, "", url);
    window.dispatchEvent(new Event(EVENT));
  };
  const clear = () => update({ engine: null, q: "", tags: [] });

  const chip = (id: string, label: string, count: number) => {
    const on = (filters.engine ?? "all") === id;
    return (
      <button key={id} onClick={() => update({ engine: id === "all" ? null : id })} aria-pressed={on}
              className={`rounded-full px-4 py-2 text-sm font-medium transition duration-300 ${
                on ? "bg-mist text-ink" : "glass text-mist/80 hover:bg-white/10 hover:text-mist"}`}>
        {label} <span className={on ? "text-ink/55" : "text-haze"}>{count}</span>
      </button>
    );
  };

  return (
    <div className="grid gap-6 lg:grid-cols-[minmax(0,1fr)_18rem] lg:gap-8">
      {/* Cột lọc: bên phải trên máy tính, trên điện thoại nằm trên lưới và thu gọn được */}
      <div className="lg:col-start-2 lg:row-start-1 lg:sticky lg:top-24 lg:self-start">
        <button type="button" onClick={() => setPanelOpen((v) => !v)} aria-expanded={panelOpen} aria-controls="bo-loc"
                className="glass inline-flex items-center gap-2.5 rounded-full px-5 py-2.5 text-sm font-semibold transition hover:bg-white/15 active:scale-95 lg:hidden">
          Bộ lọc
          {badge > 0 && <span className="grid min-w-5 place-items-center rounded-full bg-berry px-1.5 text-xs text-ink">{badge}</span>}
        </button>
        <div id="bo-loc" className={`${panelOpen ? "mt-3 block" : "hidden"} lg:mt-0 lg:block`}>
          <FilterPanel filters={filters} tagIndex={tagIndex} dirty={dirty} onChange={update} onClear={clear}
                       className="lg:max-h-[calc(100dvh-8rem)] lg:overflow-y-auto" />
        </div>
      </div>

      <div className="min-w-0 lg:col-start-1 lg:row-start-1">
        <div className="mb-6 flex flex-wrap items-center justify-between gap-x-4 gap-y-3">
          <div className="flex flex-wrap gap-2.5">
            {chip("all", "Tất cả", games.length)}
            {engines.map((e) => chip(e, e, games.filter((g) => g.engine === e).length))}
          </div>
          <p aria-live="polite" className="text-sm text-haze">{list.length} game</p>
        </div>

              {shown.map((g, i) => (
                // stagger: thẻ vào lần lượt, trần 8 nhịp để danh sách dài không phải chờ
                <div key={g.slug} className="anim-pop" style={{ "--d": `${Math.min(i, 8) * 55}ms` } as React.CSSProperties}>

              {!dirty && Array.from({ length: Math.max(0, 3 - list.length) }, (_, i) => <GhostCard key={`ghost-${i}`} />)}
            </div>
            {list.length > shown.length && (

                        className="glass rounded-full px-7 py-3 text-sm font-semibold transition duration-300 hover:-translate-y-0.5 hover:bg-white/15 active:scale-95">
                  Xem thêm ({list.length - shown.length} game)
                </button>
                <p className="text-xs text-haze">Đang hiện {shown.length}/{list.length}</p>
              </div>
            )}
          </>
        ) : (
          <div className="glass grid place-items-center rounded-[1.75rem] px-6 py-14 text-center">
            <EmptyArt />
            <p className="mt-4 font-display text-xl font-bold">Không có game nào khớp bộ lọc</p>
            <p className="mt-2 max-w-[40ch] text-sm text-haze">Thử bỏ bớt tag, đổi sang &ldquo;Hoặc&rdquo; hoặc gõ tên khác.</p>
            {dirty && (
              <button type="button" onClick={clear}
                      className="mt-6 rounded-full bg-berry px-6 py-2.5 text-sm font-semibold text-ink transition hover:-translate-y-0.5 active:scale-95">
                Xóa bộ lọc

import Cover from "./Cover";

// Khớp với src/app/search-index.json/route.ts
type Item = { slug: string; title: string; engine: string; version: string; tags: string[]; thumb: string };
type Index = { items: Item[]; fuse: Fuse<Item> };

// Tải chỉ mục + fuse.js đúng một lần, lúc người dùng sắp dùng tìm kiếm (rê chuột / focus / mở). Lỗi thì lần sau thử lại.
let pending: Promise<Index> | null = null;
function loadIndex(): Promise<Index> {
  pending ??= Promise.all([
    import("fuse.js"),
    fetch("/search-index.json").then((r) => { if (!r.ok) throw new Error(String(r.status)); return r.json() as Promise<Item[]>; }),
  ]).then(([{ default: FuseCtor }, items]) => ({
    items,
    fuse: new FuseCtor(items, { keys: ["title", "tags", "engine"], threshold: 0.35 }),
  })).catch((e) => { pending = null; throw e; });
  return pending;
}

export default function SearchBox() {
  const router = useRouter();
  const [open, setOpen] = useState(false);
  const [q, setQ] = useState("");
  const [idx, setIdx] = useState(0);
  const [index, setIndex] = useState<Index | null>(null);
  const [failed, setFailed] = useState(false);
  const results = index ? (q ? index.fuse.search(q).slice(0, 6).map((r) => r.item) : index.items.slice(0, 5)) : [];

  useEffect(() => {
    if (!open || index) return;
    let alive = true;
    setFailed(false);
    loadIndex().then((x) => { if (alive) setIndex(x); }, () => { if (alive) setFailed(true); });
    return () => { alive = false; };
  }, [open, index]);
  const warm = () => { loadIndex().catch(() => {}); };

  const close = useCallback(() => { setOpen(false); setQ(""); setIdx(0); }, []);
  const dialogRef = useModal(open, close); // khóa cuộn, nhốt Tab, Esc đóng, trả focus

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") { e.preventDefault(); setOpen(true); }
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, []);

  const onInputKey = (e: React.KeyboardEvent) => {
    if (e.key === "ArrowDown") { e.preventDefault(); setIdx((i) => Math.min(i + 1, results.length - 1)); }
    if (e.key === "ArrowUp") { e.preventDefault(); setIdx((i) => Math.max(i - 1, 0)); }
    if (e.key === "Enter" && results[idx]) { router.push(`/game/${results[idx].slug}`); close(); }
  };

  return (

      <button onClick={() => setOpen(true)} onPointerEnter={warm} onFocus={warm} aria-label="Tìm kiếm"

          <circle cx="9" cy="9" r="6" /><path d="m14 14 4 4" />
        </svg>
        <span className="hidden sm:inline">Tìm kiếm</span>
        <kbd className="hidden rounded-md bg-white/10 px-1.5 py-0.5 font-sans text-[11px] text-haze md:inline">Ctrl K</kbd>
      </button>

        <div className="anim-fade fixed inset-0 z-50 flex items-start justify-center bg-ink/70 px-4 pt-[12vh] md:backdrop-blur-sm" onClick={close}>
          <div ref={dialogRef} tabIndex={-1} role="dialog" aria-modal="true" aria-label="Tìm game"
               className="glass anim-pop w-full max-w-xl rounded-[1.75rem] p-3 shadow-2xl shadow-black/50 outline-none"
               onClick={(e) => e.stopPropagation()}>
            <input data-autofocus value={q} onChange={(e) => { setQ(e.target.value); setIdx(0); }} onKeyDown={onInputKey}
                   placeholder="Tìm theo tên, tag hoặc engine"
                   className="w-full rounded-2xl bg-white/[0.07] px-5 py-3.5 text-lg outline-none placeholder:text-haze focus:bg-white/10" />
            <ul className="mt-2 space-y-1">
              {results.map((g, i) => (
                <li key={g.slug}>
                  <Link href={`/game/${g.slug}`} onClick={close} onMouseEnter={() => setIdx(i)}
                        className={`flex items-center gap-3.5 rounded-2xl p-2 transition ${i === idx ? "bg-white/10" : ""}`}>

    .filter((g) => g.slug !== current.slug)
    .map((g) => ({
      g,
      score: g.tags.filter((t) => current.tags.includes(t)).length * 2 + (g.engine === current.engine ? 1 : 0),

    .slice(0, 4)
    .map((x) => x.g);

  if (list.length === 0) return null;

  return (
    <section>
      <h2 className="mb-3 px-1 font-display text-xl font-bold tracking-tight">Game tương tự</h2>
      <ul className="space-y-2">

import data from "./games.json";

export const MAX_EXTRA_IMAGES = 10;

export type Game = {
  slug: string;
  title: string;
  engine: string; // Ren'Py, RPGM, Unity, Unreal, Godot... tùy ý
  version: string;
  sizeGB: number;
  cover: string; // ẢNH CHÍNH: hiện ở trang chủ, thẻ game và làm nền trang game
  images: string[]; // ẢNH BỔ SUNG (tối đa 10): để hiểu thêm về game, không gồm ảnh chính
  tags: string[];
  description: string;
  updatedAt: string;
  accent?: string; // màu nhấn #rrggbb, quan_ly.py tự tính từ ảnh chính (thiếu thì web dùng màu hồng mặc định)
  accent2?: string;
  thumb?: string; // ảnh chính bản nhỏ 640px (quan_ly.py sinh): thẻ game, tìm kiếm, game tương tự. Thiếu thì dùng cover
  blur?: string; // ảnh mờ 32px dạng data URI: placeholder khi ảnh thật chưa tải xong + nền hero
  links: { label: string; url: string }[];
};

export const thumbOf = (g: Pick<Game, "thumb" | "cover">) => g.thumb || g.cover;

type Raw = Omit<Game, "cover" | "images" | "tags" | "links"> &
  Partial<Pick<Game, "cover" | "images" | "tags" | "links">>;

// Chuẩn hóa dữ liệu cũ: ảnh chính không lặp lại trong ảnh bổ sung, cắt tối đa 10.
export const games: Game[] = (data as unknown as Raw[]).map((g) => {
  const cover = g.cover || g.images?.[0] || "";
  const images = Array.from(new Set(g.images ?? []))
    .filter((u) => u && u !== cover)
    .slice(0, MAX_EXTRA_IMAGES);
  return { ...g, cover, images, tags: g.tags ?? [], links: g.links ?? [] };
});

  });
}

// Tag đặc biệt hiện thành huy hiệu trên thẻ game (không phân biệt hoa/thường/dấu):
//   "Việt hóa" -> huy hiệu "Việt hóa" cạnh engine

//   "18+"      -> nhãn đỏ góc phải dưới ảnh
const VH = new Set(["viet hoa", "viethoa", "vh"]);
const CHEAT = new Set(["cheat"]);
const ADULT = new Set(["18+"]);

export type PatchBadge = { label: string; from: string; to: string };

const TEAL = "#2dd4bf";
const AMBER = "#fbbf24";

export function gameBadges(tags: string[]): { adult: boolean; patch: PatchBadge | null } {
  const keys = tags.map(fold);
  const vh = keys.some((k) => VH.has(k));
  const cheat = keys.some((k) => CHEAT.has(k));
  const adult = keys.some((k) => ADULT.has(k));
  const patch: PatchBadge | null =
    vh && cheat ? { label: "VH&Cheat", from: TEAL, to: AMBER }
    : vh ? { label: "Việt hóa", from: TEAL, to: TEAL }
    : cheat ? { label: "Cheat", from: AMBER, to: AMBER }
    : null;
  return { adult, patch };
}

// Tag đã thể hiện bằng huy hiệu thì không lặp lại trong danh sách tag hiện khi rê chuột.
export const isBadgeTag = (tag: string) => {
  const k = fold(tag);
  return VH.has(k) || CHEAT.has(k) || ADULT.has(k);
};

import type { Game } from "@/data/games";

export type SortKey = "new" | "old" | "name" | "size";
export type TagMode = "or" | "and";

export type Filters = {
  engine: string | null; // null = tất cả engine
  q: string; // tìm theo tên game
  tags: string[]; // nhãn chuẩn (TagInfo.label)
  mode: TagMode; // or: có ít nhất 1 tag | and: có đủ mọi tag
  sort: SortKey;
};

export type TagInfo = { key: string; label: string; count: number };

export const SORTS: { key: SortKey; label: string }[] = [
  { key: "new", label: "Mới cập nhật" },
  { key: "old", label: "Cũ nhất" },
  { key: "name", label: "Tên A đến Z" },
  { key: "size", label: "Dung lượng nhỏ nhất" },
];

// Bỏ dấu + chữ thường: "Đại Hiệp" khớp "dai hiep"; "NTR" khớp "ntr".
export const fold = (s: string) =>
  s.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/đ/g, "d").trim();

// Gộp tag gợi ý (presets) với tag thật trong dữ liệu. Trùng (không phân biệt hoa/thường/dấu) thì gộp một.
// Tag có game xếp trước (nhiều game nhất lên đầu), tag chưa có game xuống cuối.
export function buildTagIndex(games: Game[], presets: string[]): TagInfo[] {
  const map = new Map<string, TagInfo>();
  for (const raw of presets) {
    const label = raw.trim();
    const key = fold(label);
    if (key && !map.has(key)) map.set(key, { key, label, count: 0 });
  }
  for (const g of games) {
    const seen = new Set<string>(); // một game ghi tag lặp thì chỉ đếm 1
    for (const raw of g.tags) {
      const label = raw.trim();
      const key = fold(label);
      if (!key || seen.has(key)) continue;
      seen.add(key);
      const hit = map.get(key);
      if (hit) hit.count++;
      else map.set(key, { key, label, count: 1 });
    }
  }
  return [...map.values()].sort((a, b) => b.count - a.count || a.label.localeCompare(b.label, "vi"));
}

// Đọc bộ lọc từ query string. Giá trị lạ (engine/tag không tồn tại, sort sai) bị bỏ, không làm hỏng trang.
export function parseFilters(search: string, engines: string[], index: TagInfo[]): Filters {
  const sp = new URLSearchParams(search);
  const engine = sp.get("engine");
  const labelByKey = new Map(index.map((t) => [t.key, t.label]));
  const tags: string[] = [];
  for (const raw of sp.getAll("tag")) {
    const label = labelByKey.get(fold(raw));
    if (label && !tags.includes(label)) tags.push(label);
  }
  const sort = sp.get("sort");
  return {
    engine: engine && engines.includes(engine) ? engine : null,
    q: sp.get("q") ?? "",
    tags,
    mode: sp.get("mode") === "and" ? "and" : "or",
    sort: SORTS.some((s) => s.key === sort) ? (sort as SortKey) : "new",
  };
}

// Giá trị mặc định không đưa lên URL cho link gọn: /?engine=Unity&tag=Romance&tag=Fantasy
export function serializeFilters(f: Filters): string {
  const sp = new URLSearchParams();
  if (f.engine) sp.set("engine", f.engine);
  if (f.q) sp.set("q", f.q);
  for (const t of f.tags) sp.append("tag", t);
  if (f.mode === "and") sp.set("mode", "and");
  if (f.sort !== "new") sp.set("sort", f.sort);
  const s = sp.toString();
  return s ? `?${s}` : "";
}

const COMPARE: Record<SortKey, (a: Game, b: Game) => number> = {
  new: (a, b) => b.updatedAt.localeCompare(a.updatedAt),
  old: (a, b) => a.updatedAt.localeCompare(b.updatedAt),
  name: (a, b) => a.title.localeCompare(b.title, "vi"),
  size: (a, b) => a.sizeGB - b.sizeGB,
};

export function applyFilters(games: Game[], f: Filters): Game[] {
  const q = fold(f.q);
  const want = f.tags.map(fold);
  const out = games.filter((g) => {
    if (f.engine && g.engine !== f.engine) return false;
    if (q && !fold(g.title).includes(q)) return false;
    if (want.length) {
      const have = new Set(g.tags.map(fold));
      const ok = f.mode === "and" ? want.every((k) => have.has(k)) : want.some((k) => have.has(k));
      if (!ok) return false;
    }
    return true;
  });
  // filter() đã tạo mảng mới nên sort tại chỗ không đụng vào mảng gốc
  return out.sort((a, b) => COMPARE[f.sort](a, b) || a.title.localeCompare(b.title, "vi"));
}

import { useEffect, useRef } from "react";

// Hộp thoại: khóa cuộn trang nền, nhốt phím Tab trong hộp, Esc để đóng, trả focus khi đóng.
// Gắn ref vào phần tử role="dialog" (cần tabIndex={-1}). Phần tử có data-autofocus được focus trước.
// Bỏ onClose nếu hộp không cho đóng bằng Esc (vd cổng 18+).
export function useModal(open: boolean, onClose?: () => void) {
  const ref = useRef<HTMLDivElement>(null);
  const closeRef = useRef(onClose);
  useEffect(() => { closeRef.current = onClose; });

  useEffect(() => {
    if (!open) return;
    const root = ref.current;
    const prevFocus = document.activeElement as HTMLElement | null;
    const prevOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";

    const focusables = () =>
      Array.from(root?.querySelectorAll<HTMLElement>('a[href],button:not([disabled]),input,[tabindex]:not([tabindex="-1"])') ?? []);
    const target = root?.hasAttribute("data-focus-root") ? root : root?.querySelector<HTMLElement>("[data-autofocus]") ?? focusables()[0] ?? root;
    target?.focus();

    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape" && closeRef.current) { e.stopPropagation(); closeRef.current(); return; }
      if (e.key !== "Tab" || !root) return;
      const f = focusables();
      if (f.length === 0) { e.preventDefault(); return; }
      const first = f[0], last = f[f.length - 1], a = document.activeElement;
      if (e.shiftKey && (a === first || !root.contains(a))) { e.preventDefault(); last.focus(); }
      else if (!e.shiftKey && (a === last || !root.contains(a))) { e.preventDefault(); first.focus(); }
    };
    document.addEventListener("keydown", onKey);
    return () => {
      document.removeEventListener("keydown", onKey);
      document.body.style.overflow = prevOverflow;
      prevFocus?.focus?.();
    };
  }, [open]);

  return ref;
}
